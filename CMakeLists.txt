cmake_minimum_required(VERSION 3.28)
project(BenScrBinary VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_BINARY_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_BINARY_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_BINARY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_binary INTERFACE)
add_library(BenScr::Binary ALIAS benscr_binary)
set_target_properties(benscr_binary PROPERTIES EXPORT_NAME Binary)
target_compile_features(benscr_binary INTERFACE cxx_std_20)
target_include_directories(benscr_binary INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_binary_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_BINARY_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()

if(BUILD_TESTING AND BENSCR_BINARY_BUILD_TESTS)
    enable_testing()
    add_executable(binary_contract_tests tests/binary_tests.cpp)
    target_link_libraries(binary_contract_tests PRIVATE BenScr::Binary)
    benscr_binary_checks(binary_contract_tests)
    add_test(NAME binary_contract COMMAND binary_contract_tests)
    add_executable(binary_header_test tests/header.cpp tests/header_peer.cpp)
    target_link_libraries(binary_header_test PRIVATE BenScr::Binary)
    benscr_binary_checks(binary_header_test)
    add_test(NAME standalone_header COMMAND binary_header_test)
endif()
if(BENSCR_BINARY_BUILD_EXAMPLES)
    add_executable(binary_example examples/binary.cpp)
    target_link_libraries(binary_example PRIVATE BenScr::Binary)
    benscr_binary_checks(binary_example)
endif()

install(TARGETS benscr_binary EXPORT BenScrBinaryTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrBinary)
configure_package_config_file(cmake/BenScrBinaryConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrBinaryConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBinary)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrBinaryConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrBinaryConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrBinaryConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBinary)
install(EXPORT BenScrBinaryTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrBinary)
